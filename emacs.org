** markdown mode
https://github.com/jrblevin/markdown-mode

#+BEGIN_SRC elisp
  (use-package markdown-mode
    :ensure t
    :custom
    (markdown-hide-markup t)
    (markdown-display-remote-images t)
    :bind
    (:map markdown-mode-map
  	("C-c C-v i" . markdown-paste-image))
    :config
    (defun markdown-paste-image ()
      "Paste an image from Wayland clipboard into current directory using `wl-paste`."
      (interactive)
      (unless (buffer-file-name)
        (user-error "Buffer must be visiting a file to determine the save directory"))
      (let* ((buffer-dir (file-name-directory (buffer-file-name)))
             (time-stamp (format-time-string "%Y%m%d_%H%M%S"))
             (filename (read-string "Image filename: " (format "image_%s.png" time-stamp)))
             (filepath (expand-file-name filename buffer-dir))
             (cmd (format "wl-paste --type image/png > %s" (shell-quote-argument filepath)))
             (exit-code (call-process-shell-command cmd)))
        (if (and (zerop exit-code) (file-exists-p filepath) (> (file-attribute-size (file-attributes filepath)) 0))
            (progn
              (insert (format "![%s](./%s)" (file-name-base filename) filename))
              (when (fboundp 'markdown-display-inline-images)
                (markdown-display-inline-images))
              (message "Saved and inserted image: %s" filename))
  	(when (file-exists-p filepath) (delete-file filepath))
  	(user-error "No image content found in Wayland clipboard"))))
    
    (defvar-local markdown--last-line nil
      "Tracks the line number where the cursor currently resides.")

    (defun markdown--line-bounds (line-num)
      "Return (beg . end) character positions for LINE-NUM, or nil if invalid."
      (when (and line-num (> line-num 0))
        (save-excursion
          (save-match-data
            (goto-char (point-min))
            (when (zerop (forward-line (1- line-num)))
              (cons (line-beginning-position) (line-end-position)))))))

    (defun markdown--unhide-current-line-matcher (limit)
      "Font-lock matcher that strips hiding properties on the cursor's line."
      (when markdown--last-line
        (let ((bounds (markdown--line-bounds markdown--last-line))
              (start (point)))
          (when bounds
            (let ((c-beg (car bounds))
                  (c-end (cdr bounds)))
              ;; Check if scanning region [start, limit] overlaps the cursor line [c-beg, c-end]
              (when (and (< start c-end) (> limit c-beg))
                (with-silent-modifications
                  (remove-text-properties (max start c-beg)
  					(min limit c-end)
  					'(display nil invisible nil composition nil))))))))
      (goto-char limit)
      nil)

    (defun markdown--update-cursor-line ()
      "Refontify old and new lines safely when the cursor moves between lines."
      (when (derived-mode-p 'markdown-mode)
        (let ((cur-line (line-number-at-pos)))
          (unless (eq cur-line markdown--last-line)
            (let ((old-line markdown--last-line))
              (setq markdown--last-line cur-line)
              ;; save-excursion and save-match-data prevent the cursor from jumping
              (save-excursion
                (save-match-data
                  ;; Re-fontify old line to re-hide its markup
                  (when old-line
                    (let ((old-bounds (markdown--line-bounds old-line)))
                      (when old-bounds
                        (font-lock-fontify-region (car old-bounds) (cdr old-bounds)))))
                  ;; Fontify new line to reveal its markup
                  (let ((new-bounds (markdown--line-bounds cur-line)))
                    (when new-bounds
                      (font-lock-fontify-region (car new-bounds) (cdr new-bounds)))))))))))

    (add-hook 'markdown-mode-hook
              (lambda ()
                (setq markdown--last-line (line-number-at-pos))
                (font-lock-add-keywords
                 nil
                 '((markdown--unhide-current-line-matcher 0 nil))
                 'append)
                (add-hook 'post-command-hook #'markdown--update-cursor-line nil t)
                (markdown-display-inline-images)))

    (set-face-attribute 'markdown-header-face-1 nil :height 2.0 :weight 'bold)
    (set-face-attribute 'markdown-header-face-2 nil :height 1.8 :weight 'bold)
    (set-face-attribute 'markdown-header-face-3 nil :height 1.6 :weight 'bold)
    (set-face-attribute 'markdown-header-face-4 nil :height 1.4 :weight 'bold)
    (set-face-attribute 'markdown-header-face-5 nil :height 1.2 :weight 'bold)
    (set-face-attribute 'markdown-header-face-6 nil :height 1.0 :weight 'bold))
